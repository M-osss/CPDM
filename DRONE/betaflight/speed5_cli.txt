# Betaflight 4.5 starting dump — speed5 6S HDZero

# Apply on a fresh F7 target after `defaults nosave`.
# This is a starting point for a 5" 2207 2550 KV / 6S / HDZero / ELRS Gem-X
# airframe. Tune filters after a hover log. Do not copy PIDs blindly onto
# another frame.

defaults nosave

# --- serial ---
serial 0 64 115200 57600 0 115200
serial 1 131073 115200 57600 0 115200
serial 5 2 115200 57600 0 115200

# UART1 = Serial RX (CRSF / ELRS)
# UART2 = MSP DisplayPort (HDZero)
# UART6 = GPS

# --- features ---
feature GPS
feature TELEMETRY
feature OSD
feature AIRMODE

# --- rx / elrs ---
set serialrx_provider = CRSF
set serialrx_inverted = OFF
set serialrx_halfduplex = OFF

# --- motors ---
set motor_pwm_protocol = DSHOT300
set dshot_bidir = ON
set dshot_bitbang = AUTO
set motor_poles = 14
set motor_idle = 5.5
set motor_output_limit = 100
set thrust_linear = 0
set throttle_boost = 0
set throttle_boost_cutoff = 15

# 2207 14-pole. If using a 12-pole motor, change motor_poles.

# --- filters (RPM first, then light RPM-harmonized) ---
set gyro_lpf1_static_hz = 0
set gyro_lpf1_dyn_min_hz = 0
set gyro_lpf1_dyn_max_hz = 0
set gyro_lpf2_static_hz = 0
set dyn_notch_count = 4
set dyn_notch_q = 500
set dyn_notch_min_hz = 80
set dyn_notch_max_hz = 600
set dterm_lpf1_dyn_min_hz = 75
set dterm_lpf1_dyn_max_hz = 150
set dterm_lpf2_static_hz = 0
set rpm_filter_harmonics = 3
set rpm_filter_fade_range_hz = 50
set rpm_filter_min_hz = 100
set rpm_filter_q = 500

# --- pid (race-ish, AOS-stiff; expect to lower D if motors run hot) ---
set p_pitch = 42
set i_pitch = 80
set d_pitch = 38
set f_pitch = 110
set p_roll = 40
set i_roll = 80
set d_roll = 36
set f_roll = 105
set p_yaw = 40
set i_yaw = 80
set d_yaw = 0
set f_yaw = 90
set d_min_roll = 24
set d_min_pitch = 26
set feedforward_averaging = 2_POINT
set feedforward_smooth_factor = 25
set feedforward_jitter_factor = 7
set feedforward_boost = 15
set tpa_rate = 70
set tpa_breakpoint = 1350
set tpa_mode = PD

# --- rates (actual, deg/s). Speed runs use low stick, not 3D. ---
rateprofile 0
set rates_type = ACTUAL
set roll_rc_rate = 18
set pitch_rc_rate = 18
set yaw_rc_rate = 18
set roll_srate = 67
set pitch_srate = 67
set yaw_srate = 67
set roll_expo = 0
set pitch_expo = 0
set yaw_expo = 0
set thr_mid = 50
set thr_expo = 0

# --- battery / 6S LiHV ---
set vbat_max_cell_voltage = 435
set vbat_full_cell_voltage = 420
set vbat_min_cell_voltage = 330
set vbat_warning_cell_voltage = 345
set vbat_scale = 110
set ibata_scale = 100
set current_meter = ESC
set battery_meter = ESC

# --- gps ---
set gps_provider = UBLOX
set gps_sbas_mode = AUTO
set gps_auto_baud = ON
set gps_ublox_use_galileo = ON
set gps_rescue_min_sats = 8
set gps_rescue_allow_arming_without_fix = ON

# --- hdzero displayport ---
set osd_displayport_device = MSP
set vcd_video_system = HD
set osd_canvas_width = 53
set osd_canvas_height = 20

# useful speed-run OSD
set osd_gps_speed_pos = 234
set osd_gps_sats_pos = 266
set osd_avg_cell_voltage_pos = 2412
set osd_link_quality_pos = 2134
set osd_rssi_dbm_pos = 2102
set osd_tim_2_pos = 2423
set osd_warnings_pos = 14697

# --- arming ---
set small_angle = 180
set gyro_cal_on_first_arm = ON
set deadband = 1
set yaw_deadband = 1

save
